   └─ Run first: ./gcp_quickstart.sh youre-edge-sports-ml

🐍 PYTHON CLIENT LIBRARY

Use gcp_client.py in your code:

from gcp_client import BigQueryClient, StorageClient, SportsMLClient

# BigQuery operations
bq = BigQueryClient()
games = bq.query_to_dataframe("SELECT * FROM sports_data.games")
bq.insert_rows("sports_data.games", [game_record])

# Cloud Storage operations
storage = StorageClient()
storage.upload_blob("youre-edge-raw-data", "local.json", "remote.json")
data = storage.download_json("youre-edge-models", "config.json")

# High-level sports ML operations
client = SportsMLClient()
client.insert_game(game_id="...", sport="NFL", ...)
client.insert_prediction(game_id="...", model_id="...", ...)
recent_games = client.get_recent_games("NFL", limit=10)
